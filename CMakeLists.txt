cmake_minimum_required(VERSION 3.25)
project(IZ1 C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

find_package(Threads REQUIRED)

add_library(battle STATIC battle.c config.c output.c proc.c)
target_compile_definitions(battle PUBLIC _POSIX_C_SOURCE=200809L)
target_compile_options(battle PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(battle PUBLIC Threads::Threads rt)

foreach(mode RANGE 0 2)
    if(mode EQUAL 0)
        set(bin tanks)
    elseif(mode EQUAL 1)
        set(bin tanks-unnamed)
    else()
        set(bin tanks-ipc)
    endif()
    add_executable(${bin} main.c)
    target_compile_definitions(${bin} PRIVATE MODE=${mode})
    target_compile_options(${bin} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${bin} PRIVATE battle)
endforeach()

add_executable(tanks-team worker.c)
target_compile_options(tanks-team PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(tanks-team PRIVATE battle)

enable_testing()
foreach(bin tanks tanks-unnamed)
    add_test(NAME ${bin}_checks COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.sh $<TARGET_FILE:${bin}>)
    set_tests_properties(${bin}_checks PROPERTIES TIMEOUT 30)
endforeach()
add_test(NAME ipc_checks COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_ipc.sh $<TARGET_FILE:tanks-ipc>)
set_tests_properties(ipc_checks PROPERTIES TIMEOUT 45)

add_library(tcp STATIC net.c)
target_compile_options(tcp PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(tcp PUBLIC battle)

add_executable(tanks-server main.c server.c)
target_compile_definitions(tanks-server PRIVATE MODE=3)
target_compile_options(tanks-server PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(tanks-server PRIVATE tcp)

foreach(bin tanks-client tanks-watch)
    add_executable(${bin} client.c)
    if(bin STREQUAL "tanks-watch")
        target_compile_definitions(${bin} PRIVATE WATCH=1)
    else()
        target_compile_definitions(${bin} PRIVATE WATCH=0)
    endif()
    target_compile_options(${bin} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${bin} PRIVATE tcp)
endforeach()

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME tcp_checks COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tcp.py
             $<TARGET_FILE:tanks-server> $<TARGET_FILE:tanks-client> $<TARGET_FILE:tanks-watch> $<TARGET_FILE:tanks>)
    set_tests_properties(tcp_checks PROPERTIES TIMEOUT 30)
endif()
