cmake_minimum_required(VERSION 3.25)
project(IZ1 C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

find_package(Threads REQUIRED)

add_library(battle STATIC battle.c config.c output.c proc.c)
target_compile_definitions(battle PUBLIC _POSIX_C_SOURCE=200809L)
target_compile_options(battle PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(battle PUBLIC Threads::Threads rt)

foreach(mode RANGE 0 2)
    if(mode EQUAL 0)
        set(bin tanks)
    elseif(mode EQUAL 1)
        set(bin tanks-unnamed)
    else()
        set(bin tanks-ipc)
    endif()
    add_executable(${bin} main.c)
    target_compile_definitions(${bin} PRIVATE MODE=${mode})
    target_compile_options(${bin} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${bin} PRIVATE battle)
endforeach()

add_executable(tanks-team worker.c)
target_compile_options(tanks-team PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(tanks-team PRIVATE battle)

enable_testing()
foreach(bin tanks tanks-unnamed)
    add_test(NAME ${bin}_checks COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.sh $<TARGET_FILE:${bin}>)
    set_tests_properties(${bin}_checks PROPERTIES TIMEOUT 30)
endforeach()
add_test(NAME ipc_checks COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_ipc.sh $<TARGET_FILE:tanks-ipc>)
set_tests_properties(ipc_checks PROPERTIES TIMEOUT 45)
